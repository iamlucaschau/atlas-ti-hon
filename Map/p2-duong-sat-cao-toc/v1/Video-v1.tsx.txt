const show = (f: number, a: number, b = Infinity) => Math.min(prog(f, a, 5), b === Infinity ? 1 : 1 - prog(f, b, 5));
const pts = (cam: GeoCam, ll: [number, number][]) => ll.map(([lo, la]) => toScreen(cam, lo, la));
/** Fraction along a screen polyline at vertex i. */
const vertexP = (ps: [number, number][], i: number) => {
  const len = (a: number, b: number) => ps.slice(a + 1, b + 1).reduce((s, q, k) => s + Math.hypot(q[0] - ps[a + k][0], q[1] - ps[a + k][1]), 0);
  return len(0, i) / len(0, ps.length - 1);
};
const count = (f: number, a: number, d: number, to: number, from = 0, digits = 0) =>
  interpolate(f, [a, a + d], [from, to], CLAMP).toLocaleString("vi-VN", { minimumFractionDigits: digits, maximumFractionDigits: digits });

/** Solid line drawn on with p, dark edge under it. */
const Track: React.FC<{ ps: [number, number][]; p: number; color?: string; w?: number; o?: number; dx?: number }> = ({
  ps, p, color = ATLAS.gold, w = 10, o = 1, dx = 0,
}) => {
  const d = ps.map(([x, y], i) => `${i ? "L" : "M"} ${x + dx} ${y}`).join(" ");
  const dash = { strokeDasharray: 1, strokeDashoffset: 1 - p };
  return (
    <svg width={1080} height={1920} style={{ position: "absolute", opacity: o }}>
      <path d={d} fill="none" stroke={ATLAS.ink} strokeWidth={w + 7} strokeLinecap="round" strokeLinejoin="round" pathLength={1} style={dash} />
      <path d={d} fill="none" stroke={color} strokeWidth={w} strokeLinecap="round" strokeLinejoin="round" pathLength={1} style={dash} />
    </svg>
  );
};

/** Rubber stamp slammed onto the screen (no camera shake: only the stamp moves). */
const Stamp: React.FC<{ text: string; x: number; y: number; f: number; at: number; out?: number }> = ({ text, x, y, f, at: a, out }) => {
  if (f < a) return null;
  const s = interpolate(f, [a, a + 6], [1.9, 1], { ...CLAMP, easing: Easing.in(Easing.cubic) });
  const w = text.length * 46 + 70;
  return (
    <svg width={1080} height={1920} style={{ position: "absolute", opacity: Math.min(prog(f, a, 3), out === undefined ? 1 : 1 - prog(f, out, 5)) }}>
      <g style={{ translate: `${x}px ${y}px`, scale: s, rotate: "-10deg" }}>
        <rect x={-w / 2} y={-50} width={w} height={100} rx={12} fill="rgba(247,242,232,0.85)" stroke={ATLAS.red} strokeWidth={9} />
        <text y={24} textAnchor="middle" fontFamily={FONT} fontWeight={800} fontSize={68} fill={ATLAS.red} letterSpacing={4}>{text}</text>
      </g>
    </svg>
  );
};

    </AbsoluteFill>
  );
};
